    HOSPITALES {
        int Id PK
        string Nombre "Efector de Salud"
        string Codigo UK "Identificador único"
        string LimiteAlmuerzo "Cierre pedidos almuerzo"
        string LimiteCena "Cierre pedidos cena"
        string LimiteAutorizacionAlmuerzo "Cierre autorizaciones"
        string LimiteAutorizacionCena "Cierre autorizaciones"
        string DietasHabilitadas "Lista dietas autorizadas"
    }

    SERVICIOS {
        int Id PK
        int HospitalId FK "Hospital pertenencia"
        string Nombre "Nombre Servicio o Sala"
        boolean VoucherIndividual "Emisión individual"
    }

    PADRON_HABILITADOS {
        int Id PK "Clave Primaria Agente"
        string DNI UK "Documento Único"
        string NombreCompleto "Apellido y Nombre"
        int RacionesMaximasDia "NUEVA: Cupo (1 o 2)"
        string OrigenAlta "NUEVA: PADRON_RRHH o EMERGENCIA"
        boolean EsGuardia24h "Flag Guardia 24h"
        boolean Activo "Habilitado para comer"
        int HospitalId FK "Hospital origen"
        int ServicioId FK "Servicio asignado"
    }

    PEDIDOS_COMIDA {
        int Id PK "Identificador Pedido"
        date FechaPedido "Día de la vianda"
        string TipoComida "'Almuerzo' | 'Cena'"
        string TipoDieta "'Normal','Gastrica',etc"
        int AgenteId FK "NUEVA: FK a PadronHabilitados"
        int ReemplazaAgenteId FK "NUEVA: FK Titular reemplazado"
        int PersonalId FK "Retrocompatibilidad (Plantel)"
        string Estado "'Pendiente' | 'Aprobado'"
        boolean EsExcepcional "Reemplazo última hora"
        string JustificacionSolicitud "Motivo o [NUTRICIÓN]"
    }

    PERSONAL {
        int Id PK "Fila de asignación"
        int AgenteId FK "NUEVA: FK a PadronHabilitados"
        string DNI "DNI del agente"
        string NombreCompleto "Nombre del agente"
        int HospitalId FK "Hospital"
        int ServicioId FK "Servicio donde figura"
        string Horario "Texto de turno"
        boolean EsGuardia24 "Guardia 24h"
        boolean ConVianda "Tiene vianda autorizada"
        boolean Activo "Estado en el servicio"
    }

    SOLICITUDES_PLANTEL {
        int Id PK "Solicitud cambio dotación"

        int ServicioId FK "Servicio solicitante"

        string Estado "'Pendiente'|'Aprobado'|'Rechazado'"
        string DatosJson "Detalle altas/bajas"

        datetime FechaRespuesta "Fecha resolución"
        int RespuestaPorId "Gerente que resolvió"
    }

Principios de Normalización Aplicados (Zero-Downtime):
  • Fuente de Verdad Única: PadronHabilitados representa a la persona física comensal con su DNI y su cupo diario exacto (RacionesMaximasDia: 1 o 2).
  • Relación Directa en Pedidos: PedidosComida.AgenteId vincula tanto los pedidos regulares de planilla como las 4 variantes de emergencia, cerrando fugas de raciones cruzadas.
  • Arquitectura Dual-Write: Las columnas legadas (PersonalId, EmergenciaNombreCompleto, EmergenciaDNI) se conservan y se siguen completando en paralelo para que el frontend actual no sufra modificaciones ni quiebres.
  • Consolidación MAX(Padrón, Personal): Si un jefe de servicio corrigió en su plantel a un agente a Guardia 24h, el sistema promueve automáticamente al agente a 2 raciones sin degradar correcciones del personal de salud.
